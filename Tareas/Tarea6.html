# 3. Relación Energy vs Danceability
energyYdanceability = df[["energy_%", "danceability_%"]]

# Creamos intervalos de 10 para Danceability
energyYdanceability["danceability_intervalo"] = pd.cut(
    energyYdanceability["danceability_%"],
    bins=range(0, 101, 10)
)

# Agrupamos por intervalo y calculamos el promedio de Energy
relacion = (
    energyYdanceability
    .groupby("danceability_intervalo", observed=True)["energy_%"]
    .mean()
    .reset_index()
)

# Gráfica
plt.figure(figsize=(10, 6))

plt.scatter(
    relacion["danceability_intervalo"].astype(str),
    relacion["energy_%"]
)

plt.xlabel("Porcentaje de danceability (%)")
plt.ylabel("Porcentaje de energy promedio (%)")
plt.title("Energy vs Danceability")

plt.show()

# 5. ¿Danceability tiene relación con los streams?

# Dividimos danceability en intervalos de 10
df["danceability_intervalo"] = pd.cut(
    df["danceability_%"],
    bins=range(0, 101, 10)
)

# Agrupamos las canciones por intervalo
# y sacamos el promedio de streams de cada grupo
streamsYdanceability = (
    df.groupby("danceability_intervalo", observed=True)["streams"]
      .mean()
      .reset_index()
)

# Importamos una librería para poder mostrar los streams en notación decimal
import matplotlib.ticker as ticker

# Graficamos
plt.figure(figsize=(10, 6))

plt.scatter(
    streamsYdanceability["danceability_intervalo"].astype(str),
    streamsYdanceability["streams"]
)

plt.xlabel("Intervalo de Danceability")
plt.ylabel("Streams promedio")
plt.title("Danceability vs. Streams")

plt.gca().yaxis.set_major_formatter(
    ticker.StrMethodFormatter('{x:,.0f}')
)

plt.show()

import pandas as pd
import matplotlib.pyplot as plt

edades = [12, 15, 18, 22, 25, 31, 35, 42, 50, 65]

    "edad": edades
})

)

grupos = df["grupo_edad"].value_counts().sort_index()

plt.bar(
    grupos.index.astype(str),
    grupos.values
)
